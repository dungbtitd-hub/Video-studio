cmake_minimum_required(VERSION 3.21)
# std::filesystem needs macOS 10.15+; iOS builds pass their own target.
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS" AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS version")
endif()
# Release builds pass -DVSTUDIO_RELEASE_VERSION=x.y.z (from the git tag).
set(VSTUDIO_RELEASE_VERSION "0.9.0" CACHE STRING "Version of the app")
project(VideoStudio VERSION ${VSTUDIO_RELEASE_VERSION} LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Only the C API is exported from shared libraries.
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
list(APPEND CMAKE_MODULE_PATH ${CMAKE_CURRENT_SOURCE_DIR}/cmake)

option(VSTUDIO_BUILD_TESTS "Build core unit tests" ON)
option(VSTUDIO_BUILD_BENCH "Build benchmarks" ON)
option(VSTUDIO_WITH_FFMPEG "Build the FFmpeg media module (desktop)" ON)
option(VSTUDIO_BUILD_CLI "Build the vstudio-cli tool" ON)
option(VSTUDIO_BUILD_DESKTOP "Build the Qt 6 desktop app" OFF)
option(VSTUDIO_WITH_LIBRAW "Build the LibRaw camera RAW module (system LibRaw, else built from source)" ON)
option(VSTUDIO_LIBRAW_FROM_SOURCE "Build LibRaw from source even when the system has it" OFF)
option(VSTUDIO_BUILD_CAPI_SHARED "Build libvstudio_capi as a shared library (Flutter FFI)" ON)

include(VStudioDeps)

if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
endif()

if(VSTUDIO_BUILD_TESTS)
  enable_testing()
endif()

add_subdirectory(core)

if(VSTUDIO_WITH_FFMPEG)
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(FFMPEG IMPORTED_TARGET libavformat libavcodec libavutil libswscale libswresample)
  endif()
  if(NOT FFMPEG_FOUND)
    # Windows: vcpkg's FFmpeg (vcpkg.json) through its CMake wrapper.
    find_package(FFMPEG QUIET)
    if(FFMPEG_FOUND)
      add_library(vstudio_ffmpeg_deps INTERFACE)
      target_include_directories(vstudio_ffmpeg_deps INTERFACE ${FFMPEG_INCLUDE_DIRS})
      target_link_directories(vstudio_ffmpeg_deps INTERFACE ${FFMPEG_LIBRARY_DIRS})
      target_link_libraries(vstudio_ffmpeg_deps INTERFACE ${FFMPEG_LIBRARIES})
      add_library(PkgConfig::FFMPEG ALIAS vstudio_ffmpeg_deps)
    endif()
  endif()
  if(FFMPEG_FOUND)
    add_subdirectory(adapters/ffmpeg)
  else()
    message(WARNING "FFmpeg development files not found: building without the FFmpeg module")
    set(VSTUDIO_WITH_FFMPEG OFF)
  endif()
endif()

if(VSTUDIO_WITH_LIBRAW)
  add_subdirectory(adapters/libraw)
endif()
if(ANDROID)
  add_subdirectory(adapters/android)
endif()
if(APPLE)
  add_subdirectory(adapters/apple)
endif()
if(WIN32)
  add_subdirectory(adapters/windows)
endif()

add_subdirectory(capi)

if(VSTUDIO_BUILD_CLI AND VSTUDIO_WITH_FFMPEG)
  add_subdirectory(tools/vstudio-cli)
endif()

if(VSTUDIO_BUILD_BENCH)
  add_subdirectory(bench)
endif()

if(VSTUDIO_BUILD_DESKTOP)
  add_subdirectory(desktop)
endif()

# ---- release packages (cpack): .deb + .tar.gz on Linux; Windows and macOS are
# packaged by .github/workflows/release.yml (NSIS installer, .dmg). See packaging/README.md.
if(VSTUDIO_BUILD_DESKTOP)
  set(CPACK_PACKAGE_NAME "videostudio")
  set(CPACK_PACKAGE_VENDOR "ITD Viet Nam")
  set(CPACK_PACKAGE_CONTACT "vanthu.itdvn@gmail.com")
  set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Video Studio - offline video editor")
  set(CPACK_PACKAGE_HOMEPAGE_URL "https://itdvietnam.com.vn")
  set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
  set(CPACK_STRIP_FILES ON)
  if(UNIX AND NOT APPLE)
    set(CPACK_GENERATOR "DEB;TGZ")
    set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)  # depends on the distribution's Qt and FFmpeg packages
    set(CPACK_DEBIAN_PACKAGE_SECTION "video")
    set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
  endif()
  include(CPack)
endif()
